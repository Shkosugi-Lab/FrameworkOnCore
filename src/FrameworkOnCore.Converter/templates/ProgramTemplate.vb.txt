Option Strict On
Option Infer On

Imports Microsoft.AspNetCore.Builder
Imports Microsoft.Extensions.DependencyInjection

' Outside the project's root namespace: the entry point is Program (StartupObject).
Namespace Global

    Public Class Program
        Public Shared Sub Main(args As String())
            ' As ASP.NET runs it on .NET Framework: in a worker process, from a copy of bin (bin stays
            ' writable), started again when the application restarts (web.config or bin changed).
            Dim exitCode As Integer
            If System.Web.Hosting.WebFormsProcess.RunInWorker(args, exitCode) Then
                System.Environment.ExitCode = exitCode
                Return
            End If

            Dim builder = WebApplication.CreateBuilder(args)
            ' ASP.NET Core's log of every request ("Request starting", "Request finished": Information) to the console, its
            ' cost on every request (IIS logs requests itself, to its files): its warnings and errors only, unless the
            ' application's configuration (appsettings.json, Logging:LogLevel:Microsoft.AspNetCore) says otherwise.
            If builder.Configuration("Logging:LogLevel:Microsoft.AspNetCore") Is Nothing Then
                Microsoft.Extensions.Logging.FilterLoggingBuilderExtensions.AddFilter(builder.Logging, "Microsoft.AspNetCore", Microsoft.Extensions.Logging.LogLevel.Warning)
            End If
            ' IIS's request log (W3C extended log files, IIS's default fields), unless web.config's httpLogging says dontLog
            ' (W3CLogging:Enabled, W3CLogging:LogDirectory in the configuration).
            builder.AddWebFormsW3CLogging()
            builder.Services.AddDistributedMemoryCache()
            builder.Services.AddSession()

            Dim app = builder.Build()

            app.UseWebFormsW3CLogging()
            ' IIS's features in front of the application (web.config's system.webServer, iis-profile.config): its response headers,
            ' URL Rewrite (UrlRewrite:Mode Iis, AspNetCore, Off), its HTTP redirection.
            app.UseWebFormsIis()

            ' IIS's default documents ("/" and "/folder/" serve default.aspx; web.config's defaultDocument), and its redirect of
            ' "/folder" to "/folder/": FrameworkOnCore's host, as IIS does them.
            app.UseSession()
            app.UseWebForms(Sub(options) options.UseAspNetCoreSessionProvider())
            app.Run()
        End Sub
    End Class

End Namespace
