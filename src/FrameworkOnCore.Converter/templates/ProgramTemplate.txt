using Microsoft.AspNetCore.Builder;
using Microsoft.Extensions.DependencyInjection;
using Microsoft.Extensions.FileProviders;

public class Program
{
    public static void Main(string[] args)
    {
        // The code pages .NET Framework had (Shift_JIS, EUC-JP, GB2312, Windows-1252, ...): .NET has only the
        // Unicode ones and ASCII, Latin-1 until a provider is registered. Before web.config is read (its
        // <globalization> encodings).
        System.Text.Encoding.RegisterProvider(System.Text.CodePagesEncodingProvider.Instance);

        // As ASP.NET runs it on .NET Framework: in a worker process, from a copy of bin (bin stays
        // writable), started again when the application restarts (web.config or bin changed).
        if (System.Web.Hosting.WebFormsProcess.RunInWorker(args, out var exitCode))
        {
            System.Environment.ExitCode = exitCode;
            return;
        }

        var builder = WebApplication.CreateBuilder(args);
        builder.Services.AddDistributedMemoryCache();
        builder.Services.AddSession();

        var app = builder.Build();

        // IIS's default document: "/" (and "/folder/") serves Default.aspx. Kestrel has no such
        // feature; the pages live in the content root, not wwwroot, so the provider is set.
        var defaults = new DefaultFilesOptions { FileProvider = new PhysicalFileProvider(builder.Environment.ContentRootPath) };
        defaults.DefaultFileNames.Clear();
        foreach (var name in new[] { "Default.aspx", "default.aspx", "Index.aspx", "index.aspx" })
        {
            defaults.DefaultFileNames.Add(name);
        }
        app.UseDefaultFiles(defaults);

        app.UseSession();
        app.UseWebForms(options => options.UseAspNetCoreSessionProvider());
        app.Run();
    }
}
