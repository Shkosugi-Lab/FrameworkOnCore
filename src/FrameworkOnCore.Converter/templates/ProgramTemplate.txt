using Microsoft.AspNetCore.Builder;
using Microsoft.Extensions.DependencyInjection;

public class Program
{
    public static void Main(string[] args)
    {
        // As ASP.NET runs it on .NET Framework: in a worker process, from a copy of bin (bin stays
        // writable), started again when the application restarts (web.config or bin changed).
        if (System.Web.Hosting.WebFormsProcess.RunInWorker(args, out var exitCode))
        {
            System.Environment.ExitCode = exitCode;
            return;
        }

        var builder = WebApplication.CreateBuilder(args);
        builder.Services.AddDistributedMemoryCache();
        builder.Services.AddSession();

        var app = builder.Build();

        // IIS's default documents ("/" and "/folder/" serve default.aspx; web.config's defaultDocument), and its redirect of
        // "/folder" to "/folder/": FrameworkOnCore's host, as IIS does them.
        app.UseSession();
        app.UseWebForms(options => options.UseAspNetCoreSessionProvider());
        app.Run();
    }
}
