       sources (see experiments/wf4c/README.md). Set up per the WebFormsForCore README. -->
  <PropertyGroup>
    <TargetFramework>net10.0</TargetFramework>
    <RootNamespace>__NAME__</RootNamespace>
    <AssemblyName>__NAME__</AssemblyName>
    <Nullable>disable</Nullable>
    <ImplicitUsings>disable</ImplicitUsings>
    <AppendTargetFrameworkToOutputPath>false</AppendTargetFrameworkToOutputPath>
    <AppendRuntimeIdentifierToOutputPath>false</AppendRuntimeIdentifierToOutputPath>
    <IntermediateOutputPath>$(BaseIntermediateOutputPath)$(Configuration)\$(TargetFramework.ToLowerInvariant())\</IntermediateOutputPath>
    <OutputType>Exe</OutputType>
    <!-- bin, as on .NET Framework: apps look for their assemblies in ~/bin (BlogEngine's extensions). -->
    <OutputPath>bin</OutputPath>
    <StartupObject>Program</StartupObject>
    <EnableDefaultContentItems>false</EnableDefaultContentItems>
    <!-- The fork (experiments/wf4c/_upstream, branch w2l/*), packed into ../_feed. -->

    <RestoreAdditionalProjectSources>$(MSBuildThisFileDirectory)..\_feed</RestoreAdditionalProjectSources>
  </PropertyGroup>

  <ItemGroup>
    <Content Remove="bin\**\*.*" />

    <Compile Remove="bin\**\*.*" />

    <Content Include="**\*.aspx;**\*.ascx;**\*.master;**\*.Master;Web.config" Exclude="bin\**" CopyToOutputDirectory="Never" />

  <ItemGroup>
    <PackageReference Include="WebFormsForCore.Web" Version="$(WebFormsForCoreVersion)" />

    <PackageReference Include="WebFormsForCore.Web.DynamicData" Version="$(WebFormsForCoreVersion)" />
  </ItemGroup>


  <!-- Assemblies .NET Framework had and .NET does not, referenced by .NET Framework packages. -->
  <ItemGroup>
    <ProjectReference Include="$(MSBuildThisFileDirectory)..\shims\System.Net.Http.WebRequest\System.Net.Http.WebRequest.csproj" />
  </ItemGroup>

  <!-- Culture data from Windows (NLS), as on .NET Framework: .NET's default ICU data differ (ja-JP's
       currency ￥ for ¥, its long date pattern), for every culture created without user overrides
       (CultureInfo.GetCultureInfo). Windows only; on Linux ICU's data are made the server's instead
       (experiments/wf4c/icu). -->
  <ItemGroup>
    <RuntimeHostConfigurationOption Include="System.Globalization.UseNls" Value="true" />
  </ItemGroup>


  <Target Name="ChangeAliasesOfNugetRefs" BeforeTargets="FindReferenceAssembliesForReferences;ResolveReferences">
    <ItemGroup>
      <ReferencePath Remove="%(Identity)" Condition="'%(FileName)' == 'System.Web' AND $([System.Text.RegularExpressions.Regex]::IsMatch(%(Identity),'[/\x5C]dotnet[/\x5C]'))" />
      <!-- CoreWCF (under WebFormsForCore.Web.Extensions) brings System.Web.Services.Description, whose
           WSDL types (WsiProfiles, ServiceDescription, ...) System.Web.Services has too (CS0433). -->
      <ReferencePath Remove="%(Identity)" Condition="'%(FileName)' == 'System.Web.Services.Description'" />
    </ItemGroup>
  </Target>
